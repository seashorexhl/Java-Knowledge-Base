## 滑动窗口 
    
    模板:
    int left = 0, right = 0;
    while (right < s.length()) {
    // 1. 扩大窗口：处理 right 指向的元素
    window.add(s[right]);
    right++;
    
        // 2. 收缩窗口：判断是否需要移动 left
        while (window.needShrink()) {
            // 处理 left 指向的元素
            window.remove(s[left]);
            left++;
        }
        // 3. 更新结果
        result = Math.max(result, right - left);
    }

## 回溯 
    
    void backtrack(路径, 选择列表) {
        if (满足结束条件) {
            result.add(路径); return;
        }
        for (选择 : 选择列表) {
            做选择(路径, 选择);
            backtrack(路径, 选择列表);
            撤销选择(路径, 选择); // ⚠️ 最容易忘的一步
        }
    }
